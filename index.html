function getResourceInfo(tags) { var service = "Community Resource"; var defaultName = "Public Community Facility"; var emoji = "📍"; if (tags.amenity === "drinking_water" || tags.amenity === "water_point") { service = "💧 Water Access"; defaultName = "Public Drinking Water Station"; emoji = "💧"; } else if (tags.amenity === "food_pantry" || tags.social_facility === "food_bank") { service = "🍞 Food Support"; defaultName = "Community Food Pantry / Bank"; emoji = "🍞"; } else if (tags.amenity === "shelter" || tags.social_facility === "shelter") { service = "🏠 Emergency Shelter"; defaultName = "Community Shelter"; emoji = "🏠"; } else if (tags.amenity === "place_of_worship") { var religion = tags.religion ? tags.religion.toLowerCase() : ""; var nameLower = tags.name ? tags.name.toLowerCase() : ""; // 1. Check Buddhism / Eastern / Chinese Temples if (religion === "buddhist" || religion === "taoist" || nameLower.includes("buddhist") || nameLower.includes("temple") || nameLower.includes("meditation") || nameLower.includes("pagoda") || nameLower.includes("zen") || nameLower.includes("dharma")) { service = "🪷 Buddhist / Eastern Temple Support"; defaultName = "Temple / Meditation Center"; emoji = "🪷"; } // 2. Check Hinduism else if (religion === "hindu" || nameLower.includes("hindu") || nameLower.includes("mandir")) { service = "🕉️ Hindu Temple Support"; defaultName = "Hindu Temple"; emoji = "🕉️"; } // 3. Check Islam else if (religion === "muslim" || nameLower.includes("mosque") || nameLower.includes("masjid") || nameLower.includes("islamic")) { service = "☪️ Mosque Support"; defaultName = "Islamic Center / Mosque"; emoji = "☪️"; } // 4. Check Sikhism else if (religion === "sikh" || nameLower.includes("gurdwara") || nameLower.includes("sikh")) { service = "☬ Gurdwara Community Kitchen"; defaultName = "Sikh Gurdwara"; emoji = "☬"; } // 5. Check Judaism else if (religion === "jewish" || nameLower.includes("synagogue") || nameLower.includes("chabad") || nameLower.includes("congregation")) { service = "✡️ Jewish Community Support"; defaultName = "Synagogue / Community Center"; emoji = "✡️"; } // 6. Check Christianity else if (religion === "christian" || nameLower.includes("church") || nameLower.includes("baptist") || nameLower.includes("chapel") || nameLower.includes("cathedral") || nameLower.includes("parish") || nameLower.includes("methodist") || nameLower.includes("lutheran") || nameLower.includes("presbyterian")) { service = "✝️ Christian Church Support"; defaultName = "Church / Christian Center"; emoji = "✝️"; } // 7. Neutral Fallback for unclassified houses of worship else { service = "🙏 Faith-Based Community Support"; defaultName = "House of Worship / Faith Center"; emoji = "🙏"; } } else if (tags.amenity === "community_centre") { service = "🤝 Community Aid Center"; defaultName = "Community Center"; emoji = "🤝"; } var name = tags.name ? (emoji + " " + tags.name) : (emoji + " " + defaultName); return { name: name, service: service }; }
